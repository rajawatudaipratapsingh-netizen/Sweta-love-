<script>
const lines = [
  "I love you, Sweta Jha ❤️",
  "You make my days brighter 🌟",
  "My heart belongs to you 💞",
  "Always yours. 💖"
];
const playBtn = document.getElementById('playBtn');
const textEl  = document.getElementById('text');

function typeText(lines, el, speed=45){
  el.style.display = 'block';
  el.textContent = '';
  let i=0,j=0;
  return new Promise(resolve=>{
    function step(){
      if(i>=lines.length){ resolve(); return; }
      const line = lines[i];
      if(j<line.length){
        el.textContent += line[j++];
        setTimeout(step, speed + Math.random()*30);
      } else {
        el.textContent += "\\n";
        i++; j=0;
        setTimeout(step, 350);
      }
    }
    step();
  });
}

function confettiBurst(){
  const canvas = document.getElementById('confetti');
  const ctx = canvas.getContext('2d');
  function resize(){
    canvas.width = canvas.offsetWidth;
    canvas.height = canvas.offsetHeight;
  }
  resize();
  const pieces = [];
  const colors = ['#ff4da6','#ffd57a','#6fb3ff','#ff66b3','#ffb3d9'];
  for(let i=0;i<70;i++){
    pieces.push({
      x: Math.random()*canvas.width,
      y: Math.random()*-canvas.height/2,
      r: (Math.random()*6)+4,
      vx:(Math.random()-0.5)*4,
      vy: Math.random()*3+2,
      c: colors[Math.floor(Math.random()*colors.length)],
      life: Math.random()*80+40,
      rot: Math.random()*360
    });
  }
  let t=0;
  function frame(){
    ctx.clearRect(0,0,canvas.width,canvas.height);
    pieces.forEach(p=>{
      ctx.save();
      ctx.translate(p.x,p.y);
      ctx.rotate(p.rot*Math.PI/180);
      ctx.fillStyle=p.c;
      ctx.fillRect(-p.r/2,-p.r/2,p.r,p.r*0.6);
      ctx.restore();
      p.x+=p.vx; p.y+=p.vy; p.rot+=p.vx*2; p.life--;
    });
    t++;
    if(t<160 && pieces.some(p=>p.life>0)) requestAnimationFrame(frame);
    else ctx.clearRect(0,0,canvas.width,canvas.height);
  }
  frame();
  window.addEventListener('resize', resize);
}

playBtn.addEventListener('click', async ()=>{
  playBtn.disabled = true;
  confettiBurst();
  await typeText(lines, textEl, 45);
  playBtn.disabled = false;
});
